\section{数学公式}

\subsection{行内公式}

矩阵乘法 $C = A \times B$ 的时间复杂度为 $O(n^3)$，空间复杂度为 $O(n^2)$。

\subsection{行间公式}

矩阵乘法的定义：

$$
C_{ij} = \sum_{k=1}^{n} A_{ik} \cdot B_{kj}
$$

行主序与列主序的等价关系：

$$
C_{\text{row}} = A_{\text{row}} \times B_{\text{row}} \quad \Longleftrightarrow \quad C_{\text{col}}^T = B_{\text{col}}^T \times A_{\text{col}}^T
$$

\subsection{对齐多行公式}

    f(x) &= ax^2 + bx + c \\
    f'(x) &= 2ax + b \\
    f''(x) &= 2a
\end{align}

\section{算法伪代码}

\begin{algorithm}[H]
    \caption{矩阵乘法朴素算法}
    \label{alg:matmul}
    \begin{algorithmic}[1]
        \REQUIRE 矩阵 $A \in \mathbb{R}^{m \times p}$, 矩阵 $B \in \mathbb{R}^{p \times n}$
        \ENSURE 矩阵 $C = A \times B \in \mathbb{R}^{m \times n}$
        \FOR{$i = 1$ \TO $m$}
            \FOR{$j = 1$ \TO $n$}
                \STATE $C_{ij} \leftarrow 0$
                \FOR{$k = 1$ \TO $p$}
                    \STATE $C_{ij} \leftarrow C_{ij} + A_{ik} \cdot B_{kj}$
                \ENDFOR
            \ENDFOR
        \ENDFOR
        \RETURN $C$
    \end{algorithmic}
\end{algorithm}
